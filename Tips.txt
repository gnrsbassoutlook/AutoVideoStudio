# ==========================
# Mac 赋予脚本可执行权限
# 使用方法：
# 1. 在访达打开本项目文件夹 AutoVideoStudio
# 2. 在文件夹空白处右键 → 新建位于文件夹位置的终端窗口
# 3. 复制下面整行，粘贴到终端回车执行
# ==========================
chmod +x *.command

# 执行完成之后，就可以直接双击 *.command 文件运行
# 如果提示"无法打开，因为来自未识别的开发者"：右键文件 → 打开，仅需要操作一次解除安全限制
# 如果报换行符错误：zsh: /Users/gnrsbass/Desktop/AutoVideoStudio/start-avs_mac.command: bad interpreter: /bin/bash^M: no such file or directory    请执行：
sed -i '' 's/\r$//' /Users/gnrsbass/Desktop/AutoVideoStudio/start-avs_mac.command
# 如果提示 sed 报错，可以换用 dos2unix（需要先安装）
# 安装dos2unix（需要brew）
brew install dos2unix
dos2unix /Users/gnrsbass/Desktop/AutoVideoStudio/start-avs_mac.command
